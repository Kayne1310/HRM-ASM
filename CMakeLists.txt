cmake_minimum_required(VERSION 3.16)
project(ChamCongTinhLuong C)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS OFF)
set(CORE src/models.c src/date_utils.c src/input.c src/employee.c src/attendance.c src/payroll.c src/storage.c)
add_library(nghiepvu STATIC ${CORE})
target_include_directories(nghiepvu PUBLIC include)
if(MSVC)
  target_compile_options(nghiepvu PRIVATE /W4 /utf-8)
else()
  target_compile_options(nghiepvu PRIVATE -Wall -Wextra -Wpedantic -Werror)
endif()
add_executable(chamcong src/main.c src/ui.c src/ui_display.c src/ui_employee.c src/ui_attendance.c src/ui_payroll.c src/search_statistics.c)
target_link_libraries(chamcong PRIVATE nghiepvu)
add_executable(test_core tests/test_core.c)
target_link_libraries(test_core PRIVATE nghiepvu)
# test_core needs a fresh empty directory; tools/test.mjs manages isolation.

add_executable(test_linked_list tests/test_linked_list.c)
target_link_libraries(test_linked_list PRIVATE nghiepvu)
